export async function loadFireEvents(
  cearaBbox,
  {
    signal,
  } = {},
) {
  const data = await fetchWfs(
    'painel_do_fogo:mv_evento_filtro',
    cearaBbox,
    db.stores.fireEvents,
    {
      signal,
    },
  );

  EventBus.emit(
    EVENTS.FIRE_EVENTS_UPDATED,
    data,
  );

  return data;
}
